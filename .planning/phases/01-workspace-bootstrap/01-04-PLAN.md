Create `.github/workflows/ci.yml` with exactly:

```yaml
name: CI

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

concurrency:
  group: ci-${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

env:
  NODE_VERSION: '20'
  PNPM_VERSION: '9.12.0'

jobs:
  build:
    name: build (tsc -b + tsup)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - run: pnpm -w build
      - name: Upload sdk dist
        uses: actions/upload-artifact@v4
        with:
          name: sdk-dist
          path: packages/sdk/dist
          retention-days: 1
          if-no-files-found: error

  lint:
    name: lint (biome ci)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - run: pnpm -w lint

  test:
    name: test (vitest)
    runs-on: ubuntu-latest
    needs: build
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - name: Download sdk dist
        uses: actions/download-artifact@v4
        with:
          name: sdk-dist
          path: packages/sdk/dist
      - run: pnpm -w test

  attw:
    name: attw (--pack)
    runs-on: ubuntu-latest
    needs: build
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - name: Download sdk dist
        uses: actions/download-artifact@v4
        with:
          name: sdk-dist
          path: packages/sdk/dist
      - run: pnpm --filter @klank/sdk exec attw --pack .

  tarball:
    name: tarball-contents check
    runs-on: ubuntu-latest
    needs: build
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - name: Download sdk dist
        uses: actions/download-artifact@v4
        with:
          name: sdk-dist
          path: packages/sdk/dist
      - run: pnpm --filter @klank/sdk run tarball:check

  changeset:
    name: changeset-required
    runs-on: ubuntu-latest
    if: github.event_name == 'pull_request'
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
      - run: pnpm install --frozen-lockfile
      - name: Verify changeset present
        run: pnpm changeset status --since=origin/${{ github.base_ref }}
```

Design notes (cross-reference RESEARCH §6 and CONTEXT D3):
- Six jobs, all required (no `continue-on-error`, no advisory tier).
- `build` uploads `packages/sdk/dist` as an artifact; `test`, `attw`, and `tarball` all `needs: build` and download the artifact — avoids rebuilding three times and proves `attw`/tarball run against the exact artifact the build job produced.
- `lint` is independent of build (biome reads source).
- `changeset` only runs on pull_request (HEAD comparison against `origin/<base_ref>`) — on push-to-main it would have nothing to compare against.
- `actions/setup-node@v4` with `cache: 'pnpm'` shares the pnpm store across jobs automatically.
- Actions pinned by version tag (v4 series). Per threat T-1-02, revisit upgrading to commit-SHA pins if a compromise is reported upstream.
- pnpm version pinned via `env.PNPM_VERSION` matching `packageManager` in root package.json.

Do NOT add any `publish` or `release` job — Phase 1 explicitly excludes publish automation (CONTEXT deferred ideas, T-1-03 disposition).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub Actions runner → npm registry | Transitive devDep install on each job |
| PR author → workflow file | Workflow edits land via PR — malicious action versions could exfiltrate secrets |
| CI job → npm publish | NOT wired this phase — publish remains manual |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-1-01 | Tampering | Transitive devDeps installed by every CI job | mitigate | Every job uses `pnpm install --frozen-lockfile` against the committed `pnpm-lock.yaml`. A drifting lockfile fails install. |
| T-1-02 | Tampering | Third-party GitHub Actions (`actions/checkout`, `actions/setup-node`, `actions/upload-artifact`, `actions/download-artifact`, `pnpm/action-setup`) | mitigate | All actions pinned to exact major tag (`@v4`). Research recommends version tags at minimum. Upgrade path to full commit SHAs documented as a future hardening step; revisit if a supply-chain incident is reported upstream. |
| T-1-03 | Elevation of Privilege | Unauthorized npm publish from CI | accept | Intentional Phase 1 scope: no `publish` job exists. Only `changeset status` runs. Publishing stays manual until a later phase ships a trusted, reviewed release workflow with scoped `NODE_AUTH_TOKEN`. |
</threat_model>
